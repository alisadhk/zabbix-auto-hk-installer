nano install_zabbix.sh

chmod +x install_zabbix.sh

./install_zabbix.sh

#!/bin/bash

read -s -p "Enter sudo password: " sudopass
echo

run_cmd() {
    echo -e "\n>> $1"
    echo "$sudopass" | sudo -S bash -c "$1"
    sleep 2
}

run_cmd "apt update && apt upgrade -y"

run_cmd "wget -4 https://repo.zabbix.com/zabbix/7.4/release/ubuntu/pool/main/z/zabbix-release/zabbix-release_latest_7.4+ubuntu24.04_all.deb"

run_cmd "dpkg -i zabbix-release_latest_7.4+ubuntu24.04_all.deb"

run_cmd "apt update"

run_cmd "apt install -y zabbix-server-mysql zabbix-frontend-php zabbix-apache-conf zabbix-sql-scripts zabbix-agent"

run_cmd "apt install -y mysql-server"

run_cmd "systemctl status mysql"

echo -e "\n>> Logging into MySQL as root..."
mysql -uroot -p <<EOF
create database zabbix character set utf8mb4 collate utf8mb4_bin;
create user zabbix@localhost identified by 'password';
grant all privileges on zabbix.* to zabbix@localhost;
set global log_bin_trust_function_creators = 1;
quit;
EOF

run_cmd "zcat /usr/share/zabbix/sql-scripts/mysql/server.sql.gz | mysql --default-character-set=utf8mb4 -uzabbix -ppassword zabbix"

echo -e "\n>> Logging into MySQL again to reset log_bin_trust_function_creators..."
mysql -uroot -p <<EOF
set global log_bin_trust_function_creators = 0;
quit;
EOF

echo -e "\n>> Updating DBPassword in zabbix_server.conf"
ZBX_CONF="/etc/zabbix/zabbix_server.conf"
echo "$sudopass" | sudo -S sed -i 's|# DBPassword=|DBPassword=password|' "$ZBX_CONF"

echo -e "\n>> Detecting local IP address:"
ip a | grep 'inet ' | grep -v '127.0.0.1' | awk '{print "http://" $2}' | cut -d/ -f1 | sed 's|$|/zabbix/|'

echo -e "\n>> Zabbix installation complete."
